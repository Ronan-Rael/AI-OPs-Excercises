package com.smartstore.billing;

import java.util.Scanner;

public class BillingApplication {

    public static boolean isNumeric(String input) {
        if (input == null || input.trim().isEmpty()) {
            return false;
        }

        try {
            double value = Double.parseDouble(input);
            return !Double.isNaN(value) && !Double.isInfinite(value);
        } catch (NumberFormatException e) {
            return false;
        }
    }

    public static double calculateDiscount(double totalAmount) {
        double discountPercentage = 0.0;

        if (totalAmount >= 1000.0 && totalAmount < 5000.0) {
            discountPercentage = 5.0;
        } else if (totalAmount >= 5000.0) {
            discountPercentage = 10.0;
        }

        return totalAmount * discountPercentage / 100.0;
    }

    public static void printBill(String customerName, String productName,
                                 double totalAmount, double discountAmount,
                                 double finalAmount) {
        double discountPercentage = 0.0;

        if (totalAmount >= 1000.0 && totalAmount < 5000.0) {
            discountPercentage = 5.0;
        } else if (totalAmount >= 5000.0) {
            discountPercentage = 10.0;
        }

        System.out.println("\n--- SmartStore Bill ---");
        System.out.println("Customer Name: " + customerName);
        System.out.println("Product Name: " + productName);
        System.out.printf("Total Amount: Rs. %.2f%n", totalAmount);
        System.out.printf("Discount Percentage: %.0f%%%n", discountPercentage);
        System.out.printf("Discount Amount: Rs. %.2f%n", discountAmount);
        System.out.printf("Final Amount: Rs. %.2f%n", finalAmount);
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter customer name: ");
        String customerName = scanner.nextLine();

        System.out.print("Enter product name: ");
        String productName = scanner.nextLine();

        System.out.print("Enter product price: ");
        String productPriceInput = scanner.nextLine();

        while (!isNumeric(productPriceInput)
                || Double.parseDouble(productPriceInput) <= 0) {
            System.out.print("Enter a valid product price (> 0): ");
            productPriceInput = scanner.nextLine();
        }

        double productPrice = Double.parseDouble(productPriceInput);

        System.out.print("Enter quantity: ");
        String quantityInput = scanner.nextLine();

        while (!isNumeric(quantityInput)
                || Double.parseDouble(quantityInput) <= 0) {
            System.out.print("Enter a valid quantity (> 0): ");
            quantityInput = scanner.nextLine();
        }

        double quantity = Double.parseDouble(quantityInput);

        double totalAmount = productPrice * quantity;
        double discountAmount = calculateDiscount(totalAmount);
        double finalAmount = totalAmount - discountAmount;

        printBill(customerName, productName, totalAmount,
                discountAmount, finalAmount);

        scanner.close();
    }
}

import java.util.Scanner;

/**
 * SmartStore billing application.

 * 10% discount for Rs. 5,000 and above.
 */
public class BillingApplication {

    public static double calculateDiscount(double totalAmount) {
        double discountPercentage = getDiscountPercentage(totalAmount);

        // Convert the percentage to the discount amount in rupees
        return totalAmount * discountPercentage / 100.0;
    }

    public static double getDiscountPercentage(double totalAmount) {
        if (totalAmount >= 1000.0 && totalAmount < 5000.0) {
            return 5.0;
        } else if (totalAmount >= 5000.0) {
            return 10.0;
        }

        return 0.0;
    }

    public static void printBill(String customerName, double totalAmount,
                                 double discountAmount, double finalAmount) {
        double discountPercentage = getDiscountPercentage(totalAmount);

        System.out.println("\n--- SmartStore Bill ---");
        System.out.println("Customer Name: " + customerName);
        System.out.printf("Total Amount: Rs. %.2f%n", totalAmount);
        System.out.printf("Discount Percentage: %.0f%%%n",
                discountPercentage);
        System.out.printf("Discount Amount: Rs. %.2f%n", discountAmount);
        System.out.printf("Final Amount: Rs. %.2f%n", finalAmount);
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter customer name: ");
        String customerName = scanner.nextLine();

        System.out.print("Enter product price: ");
        double productPrice = readPositiveNumber(
                scanner, "Enter a valid product price: ");

        System.out.print("Enter quantity: ");
        int quantity = (int) readPositiveNumber(
                scanner, "Enter a valid quantity: ");

        // Validate that the entered quantity is a whole number
        if (quantity < 1) {
            quantity = 1;
        }

        // Calculate purchase amount
        double totalAmount = productPrice * quantity;

        // Calculate discount and final amount
        double discountAmount = calculateDiscount(totalAmount);
        double finalAmount = totalAmount - discountAmount;

        printBill(customerName, totalAmount, discountAmount, finalAmount);

        scanner.close();
    }

    // Reads and validates positive numeric input
    public static double readPositiveNumber(Scanner scanner, String prompt) {
        while (!scanner.hasNextDouble()) {
            System.out.print(prompt);
            scanner.next();
        }

        double value = scanner.nextDouble();

        while (value <= 0 || Double.isInfinite(value)
                || Double.isNaN(value)) {
            System.out.print(prompt);
            while (!scanner.hasNextDouble()) {
                scanner.next();
            }
            value = scanner.nextDouble();
        }

        return value;
    }
